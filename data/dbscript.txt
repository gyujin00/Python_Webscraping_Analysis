# Database 생성

# 1.MySQL에 root 계정으로 접속한다
mysql -u root –p

# 2.현재 생성된 데이터베이스 목록을 확인한다
show databases;

# 3.기본 시스템 데이터베이스(mysql)를 사용한다
use mysql;

# 4.파이썬에서 사용할 데이터베이스(python_db)를 생성한다
create database python_db;

# 5.외부 접속이 가능한 python 사용자 계정을 생성한다
create user 'python'@'%' identified by 'python';

# 6.python_db에 대해 python 사용자에게 모든 권한을 부여한다
grant all privileges on python_db.* to 'python'@'%';

# 7.변경된 권한 정보를 즉시 반영한다
flush privileges;

# 8.생성된 사용자와 접속 가능한 host를 확인한다
select user, host from user;

# 9.root 계정에서 로그아웃한다
exit;

# 10.python 사용자로 MySQL에 접속한다, 비밀번호: python
mysql -u python -p

# 11.현재 로그인한 사용자(python)의 권한을 확인한다
show grants for current_user;

# 12.python_db 데이터베이스를 사용한다
use python_db;
